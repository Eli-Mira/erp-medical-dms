+"""Payment receipt service: create, void, and list receipts linked to invoices.
+
+Receipts are append-only. Voiding appends an AR counter-entry (adjustment debit)
+and flips the receipt status to Voided. All writes happen in the caller's
+transaction.

+def _lock_invoice(db, invoice_id):
+    row = (
+        db.execute(
+            text("SELECT * FROM sales.sales_invoice WHERE id=:id FOR UPDATE"),
+            {"id": invoice_id},
+        )
+        .mappings()
+        .one_or_none()
+    )
+    if row is None:
+        return None
+    return dict(row)

+def _lock_receipt(db, receipt_id):
+    row = (
+        db.execute(
+            text("SELECT * FROM finance.payment_receipt WHERE id=:id FOR UPDATE"),
+            {"id": receipt_id},
+        )
+        .mappings()
+        .one_or_none()
+    )
+    if row is None:
+        return None

+    return db.execute(
+        text(

+def void_receipt(db, receipt_id, actor_id, reason_id):
+    """Void a receipt: append AR adjustment debit, flip status to Voided."""
+    existing = fetch_receipt(db, receipt_id)
+    if existing is None:
+        return None
+    # All lifecycle operations lock invoice before receipt/stock, never in reverse.
+    _lock_invoice(db, existing["invoice_id"])
+    receipt = _lock_receipt(db, receipt_id)
+    if receipt is None:
+        return None
+    if receipt["status"] != "Active":
+        raise ValueError("Only Active receipts can be voided")
+    payment = (
+        db.execute(
+            text(
+                "SELECT * FROM finance.ar_entry WHERE receipt_id=:r AND entry_type='payment'"
+            ),
+            {"r": receipt_id},
+        )
+        .mappings()
+        .one_or_none()
+    )
+    if (
+        payment is None
+        or payment["customer_id"] != receipt["customer_id"]
+        or payment["invoice_id"] != receipt["invoice_id"]
+        or payment["credit"] != receipt["amount"]
+        or payment["debit"] != 0
+    ):
+        raise ValueError("Original receipt payment does not reconcile")
+
+    # AR adjustment debit (increases customer balance back)
+    finance_ar.record_entry(
+        db,
+        customer_id=receipt["customer_id"],
+        entry_type="adjustment",
+        debit=receipt["amount"],
+        credit=0,
+        on_date=db.execute(text("SELECT CURRENT_DATE")).scalar_one(),
+        reason_id=reason_id,
+        note=f"Void receipt {receipt['receipt_no']}",
+        actor_id=actor_id,
+        invoice_id=receipt["invoice_id"],
+        receipt_id=receipt_id,
+        reverses_entry_id=payment["id"],
+    )
+
+    db.execute(
+        text(
+            """
+            UPDATE finance.payment_receipt
+            SET status='Voided', voided_at=now(), voided_by=:actor,
+                void_reason_id=:reason, updated_by=:actor, updated_at=now(),
+                version=version+1
+            WHERE id=:id
+            """
+        ),
+        {"actor": actor_id, "reason": reason_id, "id": receipt_id},
+    )
+
+    return _lock_receipt(db, receipt_id)

+    """Reverse a Posted invoice atomically (append-only compensating entries):
+

+    """

+        raise ValueError("Only a Posted invoice can be reversed")

+            "WHERE invoice_id=:i AND status='Active'"

+    if active and active > 0:
+        raise ValueError("Active receipts exist; void them first")
+

+def race(life, monkeypatch, calls):
+    import app.finance.payment_router as pr
+    import app.sales.router as sr
+
+    barrier = Barrier(2)
+    # Surface actual server exceptions instead of hiding a failed race behind HTTP 500.
+    monkeypatch.setattr(life.client._transport, "raise_server_exceptions", True)
+    pids = set()
+    original = sr._invoice
+
+    def synchronized(db, user, invoice_id, *, lock=False):
+        if lock:
+            pids.add(db.execute(text("SELECT pg_backend_pid()")).scalar_one())
+            barrier.wait(timeout=10)
+        return original(db, user, invoice_id, lock=lock)
+
+    monkeypatch.setattr(sr, "_invoice", synchronized)
+    monkeypatch.setattr(pr, "_invoice", synchronized)
+    with ThreadPoolExecutor(max_workers=2) as pool:
+        futures = [pool.submit(call) for call in calls]
+        responses = [future.result(timeout=20) for future in futures]
+    assert len(pids) == 2, (
+        "Concurrency must use two independent PostgreSQL transactions"
+    )
+    return responses

+def bindings():
+    paths = {
+        p
+        for folder in ("app", "tests", "alembic", "web", "scripts")
+        for p in (ROOT / folder).rglob("*")
+        if p.is_file() and "__pycache__" not in p.parts
+    }
+    paths.update(
+        p
+        for pattern in (
+            "*.xlsx",
+            ".env*",
+            "*compose*.yml",
+            "Dockerfile",
+            "requirements.txt",
+        )
+        for p in ROOT.glob(pattern)
+        if p.is_file()
+    )
+    result = {}
+    for path in sorted(paths):
+        with path.open("rb") as source:
+            result[path.relative_to(ROOT).as_posix()] = hashlib.file_digest(
+                source, "sha256"
+            ).hexdigest()
+    return result
